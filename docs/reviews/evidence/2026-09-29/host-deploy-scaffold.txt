       python3 -m venv "/home/jinao/voc_deploy_check/.venv"
       "/home/jinao/voc_deploy_check/.venv/bin/pip" install PySide6 numpy pyserial PyYAML RPi.GPIO
  2) 放入首个 release（或直接把升级包放进 updates/ 由升级器安装）:
       ln -sfn "/home/jinao/voc_deploy_check/releases/loadport-<version>" "/home/jinao/voc_deploy_check/current"
  3) 启用并启动:
       systemctl --user enable --now voc-updater.path
       systemctl --user start voc-gui.service
       systemctl --user is-active voc-gui.service
